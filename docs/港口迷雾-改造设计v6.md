# 港口迷雾 · 改造设计 v6（代码级：怎么改、改成什么样）

目标：把“阻塞式、一辆车一辆车干”的脚本，改成“10 Hz 心跳 + 非阻塞 Actor”，一次解决：
①控制超时被系统安全停车；②泊位交付做不完整；③车辆互相顶撞；④路线脱路抄近。

## 1. 根因与对策

| 现象 | 根因 | 代码级对策 |
|---|---|---|
| 车突然不动、`visible=[]`、就位无效 | 说明 4.3：任一车 1 s 未收到控制 → 系统安全停车 + 超时制动；我们的 `while` 只喂当前车 | 心跳主循环：每 0.1 s 给四辆车各发一条指令 |
| 泊位交付永远差一步 | 站位朝向错（`[face] Car_D yaw=92 目标方位=-143`）、等级卡 lv1、精确点用错 | 泊位六步状态机 + 末段朝向 + 精确点 + P 档静置 |
| 后车顶上 A 车 | `release` 后停在原地、闸口走廊无互斥 | Car_A 不停车规则 + 走廊互斥预约 |
| 压线/脱路/撞墙 | 点连线抄近路（穿 `Obstacle_2`）、卡死踢油门 | 车道中线图 + 逐点合法性校验 + 禁倒车跨区 |

### 2.1 心跳主循环

```python
def run(api, plan, budget=600.0, dt=0.10):
    w = World(api); sched = Scheduler(w)
    tasks = sched.start(plan)
    t0 = time.time(); last = {v: 0.0 for v in w.ids}
    while True:
        now = time.time(); left = budget - (now - t0)
        if left <= 8.0: sched.wrap_up()
        w.refresh()
        cmds = {v: IDLE() for v in w.ids}
        for vid, gen in list(tasks.items()):
            if gen is None: continue
            try: c = gen.send(None)
            except StopIteration: tasks[vid] = None; continue
            except Exception as exc: sched.on_error(vid, exc); tasks[vid] = None; continue
            if c is not None: cmds[vid] = c
        cmds = sched.arbitrate(cmds)
        cmds = {v: sched.gov.fix(c, v) for v, c in cmds.items()}
        api.set_vehicle_controls(cmds)
        for v in last: last[v] = now
        if left <= 0: break
        time.sleep(dt)
    sched.final_stop()
```

### 2.2 Actor 生成器（路径跟随）

```python
def follow_path(w, vid, pts, v_cap=5.2, stop_m=2.0, stall_s=4.0):
    idx, stall_t0 = 0, None
    while idx < len(pts) - 1:
        p, yaw, v = w.pose(vid)
        idx, tgt = nearest_lookahead(pts, idx, p, dist=max(2.0, 0.9 * v + 2.0))
        if dist(p, pts[-1]) < stop_m: break
        steer = pure_pursuit(p, yaw, tgt, w.wheelbase(vid))
        thr, brk = speed_pi(v, min(v_cap, curve_speed(pts, idx)))
        if abs(v) < 0.15:
            stall_t0 = stall_t0 or time.time()
            if time.time() - stall_t0 > stall_s:
                yield from nudge_forward(w, vid); stall_t0 = None
        else:
            stall_t0 = None
        yield VehicleControl(throttle=thr, brake=brk, steering=steer)
    yield VehicleControl(throttle=0.0, brake=0.6, steering=0.0)
```

旧函数迁移表：`drive_to`/`drive_polyline`→`follow_path`；`hold_still`→`hold`；`creep_to`→`creep`；`face_target`→`face`；`back_out`/`turn_around` 保留但限速 <1.2 m/s；`drive_multi` 删除（并行改由调度器挂多个生成器）。

### 2.3 调度器互斥

```python
ZONES = {
    "mutex6": lambda x, y: abs(y + 20.22) < 6.0 and 725.0 <= x <= 732.0,
    "gate":   lambda x, y: 727.3 <= x <= 730.3 and -34.0 <= y <= -6.0,
    "berthA": lambda x, y: 620 <= x <= 646 and -72 <= y <= -55,
    "berthB": lambda x, y: 823 <= x <= 846 and -72 <= y <= -58,
}
def arbitrate(self, cmds):
    for vid, c in cmds.items():
        x, y, _, _ = self.w.pose(vid)
        d = min(self.w.gap(vid, o) for o in self.w.ids if o != vid)
        if d < 4.0: c = VehicleControl(0.0, 1.0, c.steering)
        elif d < 6.0: c = VehicleControl(0.0, max(c.brake, 0.3), c.steering)
        if ZONES["mutex6"](x, y) and not self.granted(vid):
            c = VehicleControl(0.0, 1.0, c.steering)
    return cmds
```

### 2.4 泊位六步状态机

```python
def berth_task(w, sched, berth, doer, observer, stand_doer, stand_obs, precise_fallback):
    yield from join(sched, [("follow", doer, [stand_doer]), ("follow", observer, [stand_obs])])
    yield from join(sched, [("face", doer, berth_c), ("face", observer, berth_c)])
    for _ in range(14):
        report_once(w, doer, berth); report_once(w, observer, berth)
        if level_of(w, berth) >= 3: break
        yield from tick(w, 1.5)
    precise = shared_position(w, berth) or precise_fallback
    yield from creep(w, doer, precise, tol=1.0)
    yield from park_hold(w, doer, seconds=2.5)
    for k in range(4):
        r = w.api.interact(doer, berth, "deliver")
        if getattr(r, "granted", False): return True
        if "precise berth" in (r.message or ""): yield from creep(w, doer, precise, tol=0.7)
        yield from park_hold(w, doer, seconds=1.2)
    return False
```
